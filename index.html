โปรเจควิชา Physical Computing

LANTERN

โคมไฟอัจฉริยะที่สั่งเปิด-ปิดด้วยเสียง

ไอเดียของเรา

LANTERN เป็นโคมไฟที่ฟังเสียงเราได้ แค่เรียกชื่อ “ตี๋” แล้วต่อด้วย “เปิด” หรือ “ปิด” ระบบจะสั่งรีเลย์ให้เอง หลอดไฟบ้านก็ติดหรือดับ ไม่ต้องลุกไปกดสวิตช์

ปัญหาที่เจอ

นอนอยู่บนเตียงแล้วต้องลุกไปปิดไฟ มือไม่ว่างตอนถือของ หรือคนที่เดินไปกดสวิตช์ได้ลำบาก เช่น ผู้สูงอายุ

สิ่งที่เราทำ

หลังบ้านแปลงเสียงเป็นข้อความ ถ้าได้ยินคำปลุก “ตี๋” ตามด้วยคำสั่ง จะส่งต่อให้ Arduino สั่งรีเลย์ต่อหรือตัดไฟ 220V ที่ไปหลอด

การทำงาน

ระบบมีสองฝั่ง ฝั่งสั่งงาน (ไมค์ หลังบ้าน Arduino) ใช้ไฟต่ำ ส่วนฝั่งไฟบ้าน 220V วิ่งผ่านรีเลย์ไปถึงหลอด สองฝั่งเจอกันที่รีเลย์ ซึ่งแยกไฟสองฝั่งออกจากกันด้วยขดลวดข้างใน

  1. ผู้ใช้พูดคำปลุก “ตี๋” แล้วต่อด้วย “เปิด” หรือ “ปิด”
  2. ไมโครโฟนส่งเสียงให้หลังบ้าน ซึ่งแปลงเสียงเป็นข้อความด้วย Speech-to-Text
  3. หลังบ้านเช็กว่ามีคำว่า “ตี๋” ไหม ถ้ามี ดูคำถัดไปว่าเปิดหรือปิด แล้วส่งคำสั่งให้ Arduino ทางสาย USB ถ้าไม่มีคำปลุกจะไม่ทำอะไร
  4. Arduino UNO R4 ส่ง HIGH (เปิด) หรือ LOW (ปิด) ไปที่ขา IN ของรีเลย์
  5. รีเลย์ต่อวงจร 220V หลอดไฟบ้านติด หรือตัดวงจร หลอดดับ

โค้ดหลัก

หลังบ้านฟังเสียงและเช็กคำปลุก ส่วน Arduino รอรับคำสั่งแล้วสั่งรีเลย์

หลังบ้าน (Python)

import serial
import speech_recognition as sr

arduino = serial.Serial("COM3", 9600)   # พอร์ตของ Arduino
recognizer = sr.Recognizer()

with sr.Microphone() as mic:
    while True:
        audio = recognizer.listen(mic)
        try:
            text = recognizer.recognize_google(audio, language="th-TH")
        except sr.UnknownValueError:
            continue

        if "ตี๋" not in text:          # ไม่มีคำปลุก ไม่ทำอะไร
            continue
        command = text.split("ตี๋", 1)[1]

        # เช็ก "เปิด" ก่อน เพราะคำว่า "เปิด" มี "ปิด" อยู่ข้างใน
        if "เปิด" in command:
            arduino.write(b"1")
        elif "ปิด" in command:
            arduino.write(b"0")

Arduino

const int RELAY_PIN = 7;

void setup() {
  Serial.begin(9600);              // รับคำสั่งจากหลังบ้านทาง USB
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW);     // เริ่มต้นไฟดับ
}

void loop() {
  if (Serial.available()) {
    char cmd = Serial.read();

    if (cmd == '1') digitalWrite(RELAY_PIN, HIGH);   // เปิดไฟ
    if (cmd == '0') digitalWrite(RELAY_PIN, LOW);    // ปิดไฟ
  }
}
